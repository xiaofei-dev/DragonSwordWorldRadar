local source = debug.getinfo(1, "S").source
local script_file = type(source) == "string" and source:sub(1, 1) == "@" and source:sub(2) or nil
local module_dir = script_file and script_file:match("^(.*)[/\\][^/\\]+$") or "."
local scripts_dir = module_dir:match("^(.*)[/\\][^/\\]+$") or "."
local mod_dir = scripts_dir:match("^(.*)[/\\][^/\\]+$") or "."

local report_path = mod_dir .. "\\runtime\\reports\\mole-completion-state.tsv"

local cached = {}
local canary_ok = false
local report_initialized = false
local completed_count = 0
local scan_complete = false

local function initialize_report()
    if report_initialized then
        return true
    end

    -- runtime\reports ships with the mod. If it is unavailable, fail safely.
    local file = io.open(report_path, "w")
    if not file then
        return false
    end

    file:write("MiniGameID\tCompleted\tSource\n")
    file:close()
    report_initialized = true
    return true
end

local function append_result(id, clear)
    if not initialize_report() then
        return false
    end

    local file = io.open(report_path, "a")
    if not file then
        return false
    end

    file:write(
        tostring(id), "\t",
        clear and "true" or "false", "\t",
        "DETUtil.CIsClearMiniGameInStandAlone\n"
    )
    file:close()
    return true
end

local function valid(obj)
    if obj == nil then return false end
    local ok, result = pcall(function() return obj:IsValid() end)
    return ok and result == true
end

local function query_mole(id, ctx)
    if scan_complete then
        return {
            status = "complete",
            value_type = "scan_state",
            value = "already_completed",
            fingerprint = "mole_scan:complete",
        }
    end

    if cached[id] ~= nil then
        return {
            status = "cached",
            value_type = "bool",
            value = cached[id] and "true" or "false",
            fingerprint = "mole:" .. tostring(id) .. ":" .. tostring(cached[id]),
        }
    end

    if id ~= 12001 and not canary_ok then
        return {
            status = "deferred",
            value_type = "safety_gate",
            value = "waiting_for_12001_canary",
            fingerprint = "mole:" .. tostring(id) .. ":deferred",
        }
    end

    ctx.phase("find_world_context")
    local world_context = FindFirstOf("DClientQuestSystem")
    if not valid(world_context) then
        return {
            status = "unavailable",
            value_type = "world_context",
            value = "DClientQuestSystem_not_available",
            fingerprint = "mole:" .. tostring(id) .. ":no_world_context",
        }
    end

    ctx.phase("find_detutil_cdo")
    local util = StaticFindObject("/Script/DS.Default__DETUtil")
    if not valid(util) then
        return {
            status = "unsupported",
            value_type = "uobject",
            value = "Default__DETUtil_not_found",
            fingerprint = "mole:" .. tostring(id) .. ":no_detutil",
        }
    end

    ctx.phase("find_clear_function")
    local fn = StaticFindObject("/Script/DS.DETUtil:CIsClearMiniGameInStandAlone")
    if not valid(fn) then
        return {
            status = "unsupported",
            value_type = "ufunction",
            value = "CIsClearMiniGameInStandAlone_not_found",
            fingerprint = "mole:" .. tostring(id) .. ":no_function",
        }
    end

    ctx.phase("call_clear_function")
    ctx.log("MOLE_CLEAR_QUERY_BEGIN", {
        mini_game_id = id,
        mutation = false,
        external_process = false,
    })

    local clear = fn(util, world_context, id)
    if type(clear) ~= "boolean" then
        return {
            status = "unsupported",
            value_type = type(clear),
            value = "unexpected_return_type",
            fingerprint = "mole:" .. tostring(id) .. ":bad_return",
        }
    end

    cached[id] = clear
    completed_count = completed_count + 1
    local wrote = append_result(id, clear)

    if id == 12001 then
        canary_ok = true
    end

    if completed_count >= 40 then
        scan_complete = true
        ctx.log("MOLE_COMPLETION_SCAN_COMPLETE", {
            row_count = completed_count,
            report_written = wrote,
        })
    end

    ctx.log("MOLE_CLEAR_QUERY_RETURN", {
        mini_game_id = id,
        completed = clear,
        report_written = wrote,
        completed_count = completed_count,
    })

    return {
        status = "ok",
        value_type = "bool",
        value = clear and "true" or "false",
        fingerprint = "mole:" .. tostring(id) .. ":" .. tostring(clear),
    }
end

local steps = {}
for id = 12001, 12040 do
    local mole_id = id
    steps[#steps + 1] = {
        kind = "custom",
        name = "mole_completion_" .. tostring(mole_id),
        class = "<DETUtil>",
        role = "mole_completion_state",
        purpose = "read_only_persistent_completion_query",
        cadence = 1,
        run = function(ctx)
            return query_mole(mole_id, ctx)
        end,
    }
end

return {
    id = "mole_completion_state",
    enabled = true,
    description = "One-pass read-only completion query for MiniGame_Mole_12001-12040.",
    steps = steps,
}
